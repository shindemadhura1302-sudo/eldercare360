    function renderApp() {
      renderHeaderMeta();

      const sections = {
        overview: renderOverview,
        myday: renderMyDay,
        medicine: renderMedicine,
        health: renderHealth,
        memories: renderMemories,
        garden: renderGarden,
        caregiver: renderCaregiver,
        mood: renderMood,
        privacy: renderPrivacy
      };

      sections[state.activeTab]();
    }

    document.addEventListener("click", (event) => {
      const tabBtn = event.target.closest("[data-tab]");
      if (tabBtn) {
        state.activeTab = tabBtn.dataset.tab;
        saveState();
        renderApp();
      }

      const roleBtn = event.target.closest(".role-option");
      if (roleBtn) {
        state.role = roleBtn.dataset.role;
        saveState();
        updateRoleLabel();
      }
    });

    document.getElementById("roleSelect").addEventListener("change", (event) => {
      state.role = event.target.value;
      saveState();
      renderApp();
    });

    document.getElementById("modeToggle").addEventListener("click", () => {
      state.mode = state.mode === "simple" ? "full" : "simple";
      saveState();
      renderApp();
    });

    document.getElementById("sosBtn").addEventListener("click", () => {
      state.sosStatus = "Initiated";
      state.alerts.unshift({
        id: Date.now(),
        title: "SOS action started",
        message: "This is a demonstration only. In a live configuration, a configured support channel would be used.",
        created: "Just now",
        reviewed: false,
        action: "I Need Assistance"
      });
      saveState();
      renderApp();
    });

    document.getElementById("startBtn").addEventListener("click", () => {
      landingPage.classList.add("hidden");
      appShell.classList.remove("hidden");
      state.started = true;
      saveState();
      renderApp();
    });

    document.getElementById("exploreBtn").addEventListener("click", () => {
      landingPage.classList.add("hidden");
      appShell.classList.remove("hidden");

      saveState();
      renderApp();
